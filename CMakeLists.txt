cmake_minimum_required(VERSION 3.20)
project(TarnishedToolkit VERSION 0.2.5 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
add_subdirectory(vendor/minhook)
set(IMGUI vendor/imgui)
add_library(TarnishedToolkit SHARED src/dllmain.cpp src/runtime.cpp src/ui.cpp src/overlay.cpp src/input.cpp src/horse_jump.cpp src/catalog.cpp src/storage.cpp
  ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
  ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx12.cpp)
target_include_directories(TarnishedToolkit PRIVATE src ${IMGUI} ${IMGUI}/backends)
target_compile_definitions(TarnishedToolkit PRIVATE UNICODE _UNICODE IMGUI_IMPL_WIN32_DISABLE_GAMEPAD)
target_link_libraries(TarnishedToolkit PRIVATE minhook d3d12 dxgi d3dcompiler dwmapi gdi32 bcrypt dinput8 dxguid)
set_target_properties(TarnishedToolkit PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(TarnishedToolkit PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp src/catalog.cpp)
target_include_directories(core_tests PRIVATE src)
if(MINGW)
  target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME command_validation_and_ownership COMMAND core_tests)
add_executable(storage_tests tests/storage_tests.cpp src/storage.cpp src/catalog.cpp)
target_include_directories(storage_tests PRIVATE src)
if(MINGW)
  target_link_options(storage_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME storage_roundtrip_and_corruption COMMAND storage_tests)
add_executable(attribute_tests tests/attribute_tests.cpp)
target_include_directories(attribute_tests PRIVATE src)
if(MINGW)
  target_link_options(attribute_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME attribute_planning_and_transactions COMMAND attribute_tests)
add_executable(renderer_test WIN32 tests/renderer_host.cpp)
target_link_libraries(renderer_test PRIVATE d3d12 dxgi)
if(MINGW)
  target_link_options(renderer_test PRIVATE -static -static-libgcc -static-libstdc++)
endif()

add_executable(horse_jump_tests tests/horse_jump_tests.cpp src/horse_jump.cpp)
target_include_directories(horse_jump_tests PRIVATE src)
target_compile_definitions(horse_jump_tests PRIVATE TT_HORSE_TESTS)
target_link_libraries(horse_jump_tests PRIVATE minhook)
if(MINGW)
  target_link_options(horse_jump_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME native_horse_hook_fixture COMMAND horse_jump_tests)
